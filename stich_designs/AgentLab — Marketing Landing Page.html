tools/tavily-search.ts
TypeScript 5.4 
import { defineTool } from '@agentlab/runtime';
import { z } from 'zod';

// Define an autonomous tool in clean declarative syntax
export const TavilySearch = defineTool({
  name: 'TavilySearch',
  description: 'Perform live real-time internet search queries.',
  parameters: z.object({
    query: z.string().describe('Target retrieval term'),
    maxResults: z.number().default(5),
  }),
  execute: async ({ query, maxResults }, { context }) => {
    // Automatic instrumentation & sandboxed execution
    const response = await fetch('https://api.tavily.com/search', {
      method: 'POST',
      headers: { 'Authorization': `Bearer ${context.env.TAVILY_KEY}` },
      body: JSON.stringify({ query, max_results: maxResults }),
    });
    
    return await response.json();
  },
});
